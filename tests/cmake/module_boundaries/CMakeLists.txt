cmake_minimum_required(VERSION 3.25)
project(ModuleBoundaryRegression NONE)
include("${VC_BOUNDARIES_FILE}")
add_library(vc_voxel INTERFACE)
add_library(vc_generation INTERFACE)
target_link_libraries(vc_generation INTERFACE vc_voxel)
if(VC_CASE STREQUAL "cycle")
    target_link_libraries(vc_voxel INTERFACE vc_generation)
elseif(VC_CASE STREQUAL "graphics")
    add_library(Vulkan::Vulkan INTERFACE IMPORTED)
    target_link_libraries(vc_voxel INTERFACE Vulkan::Vulkan)
elseif(VC_CASE STREQUAL "lua")
    add_library(vc_lua INTERFACE)
    target_link_libraries(vc_voxel INTERFACE vc_lua)
endif()
vc_validate_module_boundaries()
