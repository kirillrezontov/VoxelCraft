# Stage DLLs before GoogleTest discovery executes a freshly linked test binary.
# This also supports installed shared dependencies and explicit IDE targets.
function(vc_stage_runtime target)
    if(MSVC)
        target_link_options(${target} PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${PROJECT_SOURCE_DIR}/cmake/windows-utf8.manifest")
    endif()
    if(WIN32)
        file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/copy-runtime-${target}-$<CONFIG>.cmake"
            CONTENT "set(runtime_dlls \"$<TARGET_RUNTIME_DLLS:${target}>\")\nif(runtime_dlls)\n  file(COPY \${runtime_dlls} DESTINATION \"$<TARGET_FILE_DIR:${target}>\")\nendif()\n")
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -P "${CMAKE_CURRENT_BINARY_DIR}/copy-runtime-${target}-$<CONFIG>.cmake" VERBATIM)
    endif()
endfunction()
